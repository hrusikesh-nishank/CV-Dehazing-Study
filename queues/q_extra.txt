python prepare_params.py --clean_dir data/reside6k_ots/GT --depth_dir data/depth/ots --limit 1000 --v_min 50 --v_max 5000 --t_lo 0 --t_hi 1 --cct_min 5000 --cct_max 10000 --out data/params_quick_uncal.npz
